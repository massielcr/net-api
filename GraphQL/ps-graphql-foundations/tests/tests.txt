query getPiesQuery {
  pies {
    id,
    name,
    price,
    inStock
  }
}

--arguments
query getFilteredPiesQuery {
  getPiesByCategory(category: "cheesecake") {
    pies {
      id,
      name,
      price,
      inStock
    }
    count
  }
} 

--alias
query getAliasedPiesQuery {
  fruitPies: getPiesByCategory(category: "fruit") {
    pies {
      id,
      name,
      price,
      inStock
    }
    count
  },
  seasonalPies: getPiesByCategory(category: "seasonal") {
    pies {

      inStock
    }
    count
  },
  cheesecakePies: getPiesByCategory(category: "cheesecake") {

      inStock
    }
    count
  }
}

--fragments
fragment PieInfo on Pie{
  name,
  averageRating,
  inStock
}
query getfragmentedPiesQuery {
  fruitPies: getPiesByCategory(category: "fruit") {
    pies {
      ...PieInfo
    }
  }
  seasonalPies: getPiesByCategory(category: "seasonal") {
    pies {
      ...PieInfo
    }
  }
  cheesecakePies: getPiesByCategory(category: "cheesecake") {
    pies {
      ...PieInfo
    }
  }
}


--variables
query getVariablePiesQuery($category: String!) {
  getPiesByCategory(category: $category) {
    pies {
      id,
      name,
      price,
      inStock
    }
    count
  }
}
{
   "category": "cheesecake"
}
